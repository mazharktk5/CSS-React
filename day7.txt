TOPIC 1 — BrowserRouter
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  By default, React renders everything on a single page (SPA).
  When the user clicks a link, the whole page reloads — like an old website.
  We want to change URL paths (like /home, /about) WITHOUT reloading the page.

SOLUTION:
  BrowserRouter is a wrapper component from react-router-dom.
  It enables the HTML5 History API so React can change the URL bar
  without a full page reload. You wrap your ENTIRE app inside it.

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// main.jsx  (or index.js)

import React from 'react';
import ReactDOM from 'react-dom/client';
import App from './App';
import { BrowserRouter } from 'react-router-dom';

ReactDOM.createRoot(document.getElementById('root')).render(
  <BrowserRouter>
    <App />
  </BrowserRouter>
);
─────────────────────────────────────────────────────────────────────────────
  Rule: BrowserRouter goes in main.jsx — wrap App once. Never wrap it twice.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
TOPIC 2 — Routes & Route / Creating the First Route
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  We have multiple pages (Home, About, Contact).
  How do we tell React: "when the URL is /about, show the About component"?

SOLUTION:
  <Routes> is the container that holds all route definitions.
  <Route> maps a URL path to a component.
  React Router looks at the current URL and renders the matching Route.

  <Routes>                         ← container, replaces old <Switch>
    <Route path="/" element={<Home />} />
    <Route path="/about" element={<About />} />
  </Routes>

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// App.jsx

import { Routes, Route } from 'react-router-dom';

function Home()    { return <h1>Home Page</h1>;    }
function About()   { return <h1>About Page</h1>;   }
function Contact() { return <h1>Contact Page</h1>; }

function App() {
  return (
    <Routes>
      <Route path="/"        element={<Home />}    />
      <Route path="/about"   element={<About />}   />
      <Route path="/contact" element={<Contact />} />
    </Routes>
  );
}

export default App;
─────────────────────────────────────────────────────────────────────────────
  Now visit localhost:5173/about → you see "About Page" without any reload.

TOPIC 3 — Link
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  If we use a normal <a href="/about"> tag, the browser reloads the entire page.
  The whole React app resets — all state is lost.

SOLUTION:
  React Router provides a <Link to="/about"> component.
  It renders as an <a> tag but intercepts the click and uses
  client-side navigation — NO page reload, state is preserved.

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// Navbar.jsx

import { Link } from 'react-router-dom';

function Navbar() {
  return (
    <nav style={{ display: 'flex', gap: '20px', padding: '10px', background: '#333' }}>
      <Link to="/"        style={{ color: 'white', textDecoration: 'none' }}>Home</Link>
      <Link to="/about"   style={{ color: 'white', textDecoration: 'none' }}>About</Link>
      <Link to="/contact" style={{ color: 'white', textDecoration: 'none' }}>Contact</Link>
    </nav>
  );
}

export default Navbar;
─────────────────────────────────────────────────────────────────────────────
  Then add <Navbar /> at the top of App.jsx above <Routes>.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
TOPIC 4 — NavLink
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  With <Link>, we can navigate — but how do we highlight the ACTIVE link
  in the navbar so the user knows which page they are on?

SOLUTION:
  <NavLink> is a smarter version of <Link>.
  It automatically adds an "active" class to the link whose path matches
  the current URL. You can style the active link differently.

  You can use the className prop with a function:
    className={({ isActive }) => isActive ? 'active-link' : 'link'}

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// Navbar.jsx

import { NavLink } from 'react-router-dom';

const linkStyle = {
  color: 'white',
  textDecoration: 'none',
  padding: '8px 12px',
  borderRadius: '4px',
};

const activeStyle = {
  ...linkStyle,
  backgroundColor: '#0077ff',
  fontWeight: 'bold',
};

function Navbar() {
  return (
    <nav style={{ display: 'flex', gap: '10px', padding: '10px', background: '#222' }}>
      <NavLink
        to="/"
        end
        style={({ isActive }) => isActive ? activeStyle : linkStyle}
      >
        Home
      </NavLink>

      <NavLink
        to="/about"
        style={({ isActive }) => isActive ? activeStyle : linkStyle}
      >
        About
      </NavLink>

      <NavLink
        to="/contact"
        style={({ isActive }) => isActive ? activeStyle : linkStyle}
      >
        Contact
      </NavLink>
    </nav>
  );
}

export default Navbar;
─────────────────────────────────────────────────────────────────────────────
  NOTE: Use "end" on the "/" route so it doesn't stay active on /about too.

TOPIC 5 — useNavigate
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  <Link> and <NavLink> work when the user clicks.
  But what if we want to navigate PROGRAMMATICALLY — for example,
  after a form is submitted, or after a login succeeds?
  We can't put a <Link> inside an onClick function.

SOLUTION:
  useNavigate() is a hook that returns a navigate function.
  Call navigate('/path') anywhere in your code — after a button click,
  after an API call, inside a useEffect, etc.

  navigate('/home')       → go to /home
  navigate(-1)            → go back (like browser back button)
  navigate(1)             → go forward
  navigate('/login', { replace: true }) → navigate and replace history

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// LoginPage.jsx

import { useState } from 'react';
import { useNavigate } from 'react-router-dom';

function LoginPage() {
  const [username, setUsername] = useState('');
  const navigate = useNavigate();

  const handleLogin = () => {
    if (username === 'admin') {
      // After successful login, redirect to dashboard
      navigate('/dashboard');
    } else {
      alert('Wrong username! Try: admin');
    }
  };

  return (
    <div style={{ padding: '40px' }}>
      <h2>Login</h2>
      <input
        type="text"
        placeholder="Enter username"
        value={username}
        onChange={(e) => setUsername(e.target.value)}
        style={{ padding: '8px', marginRight: '10px' }}
      />
      <button onClick={handleLogin} style={{ padding: '8px 16px' }}>
        Login
      </button>
    </div>
  );
}

export default LoginPage;
─────────────────────────────────────────────────────────────────────────────
  Type "admin" → click Login → navigates to /dashboard automatically.

TOPIC 6 — Nested Routes & Outlet
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  In a dashboard, the sidebar/header stays the same, but only the
  main content area changes based on the URL (/dashboard/profile,
  /dashboard/settings). How do we share a layout between child routes?

SOLUTION:
  Nested Routes let a parent Route contain child Routes inside it.
  The parent component uses <Outlet /> as a placeholder — React Router
  renders the matching child component inside that Outlet.

  Parent:  /dashboard      → renders DashboardLayout (with sidebar)
  Child:   /dashboard/profile   → Outlet renders <Profile />
  Child:   /dashboard/settings  → Outlet renders <Settings />

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// App.jsx

import { Routes, Route } from 'react-router-dom';
import DashboardLayout from './DashboardLayout';
import Profile from './Profile';
import Settings from './Settings';
import Home from './Home';

function App() {
  return (
    <Routes>
      <Route path="/" element={<Home />} />

      {/* Parent route — DashboardLayout always renders */}
      <Route path="/dashboard" element={<DashboardLayout />}>
        {/* Child routes render inside <Outlet /> */}
        <Route path="profile"  element={<Profile />}  />
        <Route path="settings" element={<Settings />} />
      </Route>
    </Routes>
  );
}

export default App;

─────────────────────────────────────────────────────────────────────────────
// DashboardLayout.jsx

import { Outlet, Link } from 'react-router-dom';

function DashboardLayout() {
  return (
    <div style={{ display: 'flex' }}>
      {/* Sidebar — always visible */}
      <aside style={{ width: '200px', background: '#f0f0f0', padding: '20px', minHeight: '100vh' }}>
        <h3>Dashboard</h3>
        <Link to="profile"  style={{ display: 'block', marginBottom: '10px' }}>Profile</Link>
        <Link to="settings" style={{ display: 'block' }}>Settings</Link>
      </aside>

      {/* Main content — child route renders here */}
      <main style={{ padding: '20px', flex: 1 }}>
        <Outlet />
      </main>
    </div>
  );
}

export default DashboardLayout;

─────────────────────────────────────────────────────────────────────────────
// Profile.jsx
function Profile() { return <h2>My Profile Page</h2>; }
export default Profile;

// Settings.jsx
function Settings() { return <h2>Settings Page</h2>; }
export default Settings;
─────────────────────────────────────────────────────────────────────────────
  Visit /dashboard/profile → sidebar shows, Profile renders in Outlet.
  Visit /dashboard/settings → sidebar stays, Settings renders in Outlet.

TOPIC 7 — Dynamic Routing (:id) & useParams
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  We have 100 products. We don't want to create 100 routes manually.
  /product/1, /product/2, /product/3 ... /product/100
  How do we handle a variable part of the URL?

SOLUTION:
  Use a dynamic segment with a colon: path="/product/:id"
  The :id is a URL parameter — it matches any value.
  Inside the component, use the useParams() hook to read that value.

  path="/product/:id"
    /product/1    → params.id = "1"
    /product/42   → params.id = "42"
    /product/abc  → params.id = "abc"

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// App.jsx — add this route

import ProductList from './ProductList';
import ProductDetail from './ProductDetail';

// Inside <Routes>:
<Route path="/products"      element={<ProductList />}   />
<Route path="/products/:id"  element={<ProductDetail />} />

─────────────────────────────────────────────────────────────────────────────
// ProductList.jsx

import { Link } from 'react-router-dom';

const products = [
  { id: 1, name: 'Laptop' },
  { id: 2, name: 'Phone' },
  { id: 3, name: 'Headphones' },
];

function ProductList() {
  return (
    <div style={{ padding: '20px' }}>
      <h2>All Products</h2>
      {products.map((p) => (
        <div key={p.id} style={{ marginBottom: '10px' }}>
          <Link to={`/products/${p.id}`}>{p.name}</Link>
        </div>
      ))}
    </div>
  );
}

export default ProductList;

─────────────────────────────────────────────────────────────────────────────
// ProductDetail.jsx

import { useParams, Link } from 'react-router-dom';

const products = [
  { id: 1, name: 'Laptop',      price: '$999',  desc: 'Fast and powerful laptop.' },
  { id: 2, name: 'Phone',       price: '$699',  desc: 'Latest smartphone model.'  },
  { id: 3, name: 'Headphones',  price: '$199',  desc: 'Noise cancelling audio.'    },
];

function ProductDetail() {
  const { id } = useParams();                             // read :id from URL
  const product = products.find((p) => p.id === Number(id));

  if (!product) return <h2>Product not found!</h2>;

  return (
    <div style={{ padding: '40px' }}>
      <h2>{product.name}</h2>
      <p>Price: {product.price}</p>
      <p>{product.desc}</p>
      <Link to="/products">← Back to Products</Link>
    </div>
  );
}

export default ProductDetail;
─────────────────────────────────────────────────────────────────────────────
  Click "Laptop" → URL becomes /products/1 → useParams gives { id: "1" }.

TOPIC 8 — 404 Page (path="*")
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  A user types a URL that doesn't exist — like /blabla or /xyz123.
  Currently nothing renders — just a blank page.
  We need to show a friendly "Page Not Found" message.

SOLUTION:
  Add a catch-all Route with path="*" at the END of your <Routes>.
  The * matches any URL that didn't match any previous route.
  Always put it LAST so it only catches unmatched paths.

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// NotFound.jsx

import { Link } from 'react-router-dom';

function NotFound() {
  return (
    <div style={{ textAlign: 'center', padding: '80px' }}>
      <h1 style={{ fontSize: '80px', margin: '0' }}>404</h1>
      <h2>Oops! Page Not Found</h2>
      <p>The page you are looking for does not exist.</p>
      <Link
        to="/"
        style={{
          display: 'inline-block',
          padding: '10px 20px',
          background: '#0077ff',
          color: 'white',
          borderRadius: '6px',
          textDecoration: 'none',
          marginTop: '20px',
        }}
      >
        Go Back Home
      </Link>
    </div>
  );
}

export default NotFound;

─────────────────────────────────────────────────────────────────────────────
// App.jsx — add * route LAST

import NotFound from './NotFound';

function App() {
  return (
    <Routes>
      <Route path="/"       element={<Home />}    />
      <Route path="/about"  element={<About />}   />
      <Route path="/contact" element={<Contact />} />

      {/* ALWAYS last — catches everything else */}
      <Route path="*" element={<NotFound />} />
    </Routes>
  );
}
─────────────────────────────────────────────────────────────────────────────
  Visit /anything-random → shows the 404 page with a link back to Home.

TOPIC 9 — Protected Routes using Navigate
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  Some pages (like Dashboard) should only be visible to logged-in users.
  If a non-logged-in user visits /dashboard, they should be redirected to /login.
  How do we guard routes based on authentication?

SOLUTION:
  Create a ProtectedRoute component that:
    1. Checks if the user is logged in
    2. If YES → renders the children (the protected page)
    3. If NO  → redirects to /login using <Navigate to="/login" />

  <Navigate> is a component (not a hook) that immediately redirects
  when rendered. It's like calling navigate() but as JSX.

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// ProtectedRoute.jsx

import { Navigate } from 'react-router-dom';

function ProtectedRoute({ children }) {
  const isLoggedIn = localStorage.getItem('user'); // simple auth check

  if (!isLoggedIn) {
    return <Navigate to="/login" replace />;
  }

  return children;
}

export default ProtectedRoute;

─────────────────────────────────────────────────────────────────────────────
// App.jsx

import ProtectedRoute from './ProtectedRoute';
import Dashboard from './Dashboard';
import Login from './Login';

function App() {
  return (
    <Routes>
      <Route path="/"      element={<Home />}    />
      <Route path="/login" element={<Login />}   />

      {/* Wrap protected routes */}
      <Route
        path="/dashboard"
        element={
          <ProtectedRoute>
            <Dashboard />
          </ProtectedRoute>
        }
      />

      <Route path="*" element={<NotFound />} />
    </Routes>
  );
}

─────────────────────────────────────────────────────────────────────────────
// Login.jsx — simple demo

import { useNavigate } from 'react-router-dom';

function Login() {
  const navigate = useNavigate();

  const handleLogin = () => {
    localStorage.setItem('user', 'loggedIn');  // simulate login
    navigate('/dashboard');
  };

  return (
    <div style={{ padding: '40px' }}>
      <h2>Login Page</h2>

TOPIC 10 — What is Prop Drilling?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  You have a deeply nested component tree:
    App → Navbar → UserAvatar

  UserAvatar needs to know the logged-in user's name.
  The user data lives in App. You pass it as a prop:
    App passes user → Navbar (Navbar doesn't need it, just passing it down)
    Navbar passes user → UserAvatar (finally used here)

  This is PROP DRILLING — passing props through components that don't
  need them just to reach a deeply nested component. It becomes messy
  and hard to maintain when the tree is 5-10 levels deep.

DEMONSTRATION OF THE PROBLEM:
─────────────────────────────────────────────────────────────────────────────
// Prop drilling — BAD pattern for deep trees

function App() {
  const user = { name: 'Ahmed' };
  return <Navbar user={user} />;
}

function Navbar({ user }) {
  // Navbar does NOT use user — just passing it down
  return <UserAvatar user={user} />;
}

function UserAvatar({ user }) {
  // Finally used here, 2 levels deep
  return <p>Hello, {user.name}</p>;
}
─────────────────────────────────────────────────────────────────────────────
  Imagine 5-6 levels deep — every component must accept and pass the prop.
  This is the problem Context API solves.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
TOPIC 11 — createContext(), Provider, value prop, useContext()
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  We need a way to share data (like user info, theme, language) globally
  across many components — without manually passing props at every level.

SOLUTION — The 4 steps of Context API:

  STEP 1: createContext()
    Creates a "storage box" that can hold global data.
    const UserContext = createContext();

  STEP 2: Provider
    Wraps the components that need access to the data.
    Everything inside the Provider can read the data.
    <UserContext.Provider value={user}>
      <App />
    </UserContext.Provider>

  STEP 3: value prop
    The data you want to share goes in the value prop.
    value can be anything: a string, object, array, function.

  STEP 4: useContext()
    Any component inside the Provider reads the data with useContext().
    const user = useContext(UserContext);
    No prop drilling needed at all.

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// UserContext.js — create the context

TOPIC 12 — Multiple Contexts (UserContext & ProductContext)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  Our app needs to share MULTIPLE pieces of global data:
  - User information (name, role, isLoggedIn)
  - Products data (product list, cart)

SOLUTION:

  Each component only subscribes to the context it needs.

    <ProductContext.Provider value={productData}>
      <App />
    </ProductContext.Provider>
  </UserContext.Provider>

TOPIC 13 — Practical Example: Context API with Toggle Theme
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEM:
  We want a Dark/Light theme toggle that affects the ENTIRE app.
  Many components need to read the current theme and update when it changes.
  Passing theme and toggleTheme as props everywhere is impractical.

SOLUTION:
  Store the theme and the toggle function in a Context.
  Provide them at the top level. Any component reads and updates them freely.

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// ThemeContext.js

import { createContext, useState, useContext } from 'react';

const ThemeContext = createContext();

// Custom provider component — keeps things organized
export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');

  const toggleTheme = () => {
    setTheme((prev) => (prev === 'light' ? 'dark' : 'light'));
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

// Custom hook for convenience
export function useTheme() {
  return useContext(ThemeContext);
}

TOPIC 14 — Sign Up & Login with Email and Password

PROBLEM:
  We want users to create an account with email + password and then
  sign in with those credentials — stored securely in Firebase.

SOLUTION:
  Firebase Auth provides two async functions:
    createUserWithEmailAndPassword(auth, email, password) → Sign Up

PRACTICAL CODE EXAMPLE:
─────────────────────────────────────────────────────────────────────────────
// SignUp.jsx

import { useState } from 'react';
import { auth } from './firebase';
import { createUserWithEmailAndPassword } from 'firebase/auth';

function SignUp() {
  const [email,    setEmail]    = useState('');
  const [password, setPassword] = useState('');
  const [message,  setMessage]  = useState('');

  const handleSignUp = async () => {
    try {
      const result = await createUserWithEmailAndPassword(auth, email, password);
      setMessage(`Account created! Welcome ${result.user.email}`);
    } catch (error) {
      setMessage(`Error: ${error.message}`);
    }
  };

  const inputStyle = {
    display: 'block', padding: '10px', margin: '10px 0',
    width: '300px', fontSize: '16px',
  };

  return (
    <div style={{ padding: '40px' }}>
      <h2>Create Account</h2>
      <input
        type="email"
        placeholder="Email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        style={inputStyle}
      />
      <input
        type="password"
        placeholder="Password (min 6 chars)"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
        style={inputStyle}
      />
      <button
        onClick={handleSignUp}
        style={{ padding: '10px 30px', background: '#28a745', color: 'white', border: 'none', borderRadius: '6px', fontSize: '16px', cursor: 'pointer' }}
      >
        Sign Up
      </button>
      {message && <p style={{ marginTop: '15px', color: message.startsWith('Error') ? 'red' : 'green' }}>{message}</p>}
    </div>
  );
}

export default SignUp;

// src/contexts/AuthContext.jsx
// Stores the current Firebase user globally using Context API

import { createContext, useContext, useEffect, useState } from 'react';
import { auth } from '../firebase';
import { onAuthStateChanged } from 'firebase/auth';

const AuthContext = createContext();

export function AuthProvider({ children }) {
  const [currentUser, setCurrentUser] = useState(null);
  const [loading,     setLoading]     = useState(true); // wait for Firebase

  useEffect(() => {
    // Firebase calls this whenever auth state changes (login/logout)
    const unsubscribe = onAuthStateChanged(auth, (user) => {
      setCurrentUser(user);
      setLoading(false);
    });

    return unsubscribe; // cleanup on unmount
  }, []);

  const value = { currentUser };

  // Don't render children until Firebase has checked auth state
  if (loading) return <p style={{ textAlign: 'center', marginTop: '100px' }}>Loading...</p>;

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  return useContext(AuthContext);
}

// src/components/ProtectedRoute.jsx

import { Navigate } from 'react-router-dom';
import { useAuth } from '../contexts/AuthContext';

function ProtectedRoute({ children }) {
  const { currentUser } = useAuth();

  if (!currentUser) {
    return <Navigate to="/login" replace />;
  }

  return children;
}

export default ProtectedRoute;

// src/pages/SignUp.jsx

import { useState } from 'react';
import { auth } from '../firebase';
import { createUserWithEmailAndPassword } from 'firebase/auth';
import { useNavigate, Link } from 'react-router-dom';

function SignUp() {
  const [email,    setEmail]    = useState('');
  const [password, setPassword] = useState('');
  const [error,    setError]    = useState('');
  const navigate = useNavigate();

  const handleSignUp = async (e) => {
    e.preventDefault();
    setError('');
    try {
      await createUserWithEmailAndPassword(auth, email, password);
      navigate('/dashboard');
    } catch (err) {
      setError(err.message);
    }
  };

  return (
    <div style={pageStyle}>
      <div style={cardStyle}>
        <h2 style={{ marginBottom: '24px' }}>Create Account</h2>
        <form onSubmit={handleSignUp}>
          <input
            type="email"
            placeholder="Email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            required
            style={inputStyle}
          />
          <input
            type="password"
            placeholder="Password (min 6 chars)"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            required
            style={inputStyle}
          />
          {error && <p style={{ color: 'red', fontSize: '14px' }}>{error}</p>}
          <button type="submit" style={btnStyle('#28a745')}>
            Sign Up
          </button>
        </form>
        <p style={{ marginTop: '16px' }}>
          Already have an account? <Link to="/login">Login</Link>
        </p>
      </div>
    </div>
  );
}

export default SignUp;

// src/pages/Login.jsx

import { useState } from 'react';
import { auth, provider } from '../firebase';
import {
  signInWithEmailAndPassword,
  signInWithPopup,
} from 'firebase/auth';
import { useNavigate, Link } from 'react-router-dom';

function Login() {
  const [email,    setEmail]    = useState('');
  const [password, setPassword] = useState('');
  const [error,    setError]    = useState('');
  const navigate = useNavigate();

  const handleEmailLogin = async (e) => {
    e.preventDefault();
    setError('');
    try {
      await signInWithEmailAndPassword(auth, email, password);
      navigate('/dashboard');
    } catch {
      setError('Invalid email or password.');
    }
  };

  const handleGoogleLogin = async () => {
    setError('');
    try {
      await signInWithPopup(auth, provider);
      navigate('/dashboard');
    } catch (err) {
      setError(err.message);
    }
  };

  return (
    <div style={pageStyle}>
      <div style={cardStyle}>
        <h2 style={{ marginBottom: '24px' }}>Login</h2>
        <form onSubmit={handleEmailLogin}>
          <input
            type="email"
            placeholder="Email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            required
            style={inputStyle}
          />
          <input
            type="password"
            placeholder="Password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            required
            style={inputStyle}
          />
          {error && <p style={{ color: 'red', fontSize: '14px' }}>{error}</p>}
          <button type="submit" style={btnStyle('#0077ff')}>
            Login with Email
          </button>
        </form>

        <div style={{ margin: '16px 0', textAlign: 'center', color: '#aaa' }}>
          — OR —
        </div>

        <button onClick={handleGoogleLogin} style={googleBtnStyle}>
          <img
            src="https://www.gstatic.com/firebasejs/ui/2.0.0/images/auth/google.svg"
            alt="G"
            width="20"
          />
          Continue with Google
        </button>

        <p style={{ marginTop: '16px' }}>
          No account? <Link to="/signup">Sign Up</Link>
        </p>
      </div>
    </div>
  );
}

export default Login;

// src/pages/Dashboard.jsx

import { auth } from '../firebase';
import { signOut } from 'firebase/auth';
import { useAuth } from '../contexts/AuthContext';
import { useNavigate } from 'react-router-dom';

function Dashboard() {
  const { currentUser } = useAuth();
  const navigate        = useNavigate();

  const handleLogout = async () => {
    await signOut(auth);
    navigate('/login');
  };

  return (
    <div style={{ minHeight: '100vh', background: '#f0f4ff', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center' }}>
      <div style={cardStyle}>
        <h2 style={{ marginBottom: '8px' }}>Dashboard</h2>
        <p style={{ color: '#555', marginBottom: '24px' }}>Welcome back!</p>

        <div style={{ background: '#e8f0fe', borderRadius: '8px', padding: '16px', marginBottom: '24px' }}>
          <p style={{ margin: 0 }}><strong>Email:</strong> {currentUser.email}</p>
          {currentUser.displayName && (
            <p style={{ margin: '8px 0 0' }}><strong>Name:</strong> {currentUser.displayName}</p>
          )}
        </div>

        <button
          onClick={handleLogout}
          style={{ ...btnStyle('#e53935'), width: '100%' }}
        >
          Logout
        </button>
      </div>
    </div>
  );
}

export default Dashboard;

// src/App.jsx

import { Routes, Route, Navigate } from 'react-router-dom';
import { AuthProvider } from './contexts/AuthContext';
import ProtectedRoute from './components/ProtectedRoute';
import Login      from './pages/Login';
import SignUp     from './pages/SignUp';
import Dashboard  from './pages/Dashboard';

function App() {
  return (
    <AuthProvider>
      <Routes>
        <Route path="/"        element={<Navigate to="/login" />} />
        <Route path="/login"   element={<Login />}   />
        <Route path="/signup"  element={<SignUp />}  />
        <Route
          path="/dashboard"
          element={
            <ProtectedRoute>
              <Dashboard />
            </ProtectedRoute>
          }
        />
      </Routes>
    </AuthProvider>
  );
}

export default App;

================================================================================
                           FULL FLOW SUMMARY
================================================================================

  1. User visits /dashboard
     → ProtectedRoute checks AuthContext
     → currentUser is null (not logged in)
     → Redirected to /login

  2. User fills email + password → clicks "Login with Email"
     → Firebase signInWithEmailAndPassword called
     → onAuthStateChanged fires → currentUser set in AuthContext
     → navigate('/dashboard') called

  3. User visits /dashboard again
     → ProtectedRoute checks AuthContext → currentUser exists
     → Dashboard renders, shows user email

  4. User clicks "Continue with Google"
     → Firebase opens Google popup
     → User selects Google account
     → onAuthStateChanged fires → currentUser set
     → navigate('/dashboard')

  5. User clicks Logout
     → Firebase signOut called
     → onAuthStateChanged fires → currentUser = null
     → navigate('/login')
     → /dashboard now redirects back to /login


================================================================================
                     QUICK REFERENCE CHEATSHEET
================================================================================

  REACT ROUTER:
  ─────────────────────────────────────────────────────────────────────────
  BrowserRouter         Wrap in main.jsx — enables client-side routing
  <Routes>              Container for all <Route> elements
  <Route path element>  Maps URL to component
  <Link to>             Navigation without page reload
  <NavLink to>          Like Link + auto active class/style
  useNavigate()         Programmatic navigation
  <Outlet />            Renders matched child route in parent layout
  :id in path           Dynamic URL segment
  useParams()           Reads :id from the URL
  path="*"              Catch-all — 404 page, always put LAST
  <Navigate to>         Immediate redirect when rendered

  CONTEXT API:
  ─────────────────────────────────────────────────────────────────────────
  createContext()        Creates the context object
  <Context.Provider>     Wraps components, shares value prop
  useContext(Context)    Reads the value in any child component

  FIREBASE AUTH:
  ─────────────────────────────────────────────────────────────────────────
  createUserWithEmailAndPassword   Sign up new user
  signInWithEmailAndPassword       Login existing user
  signInWithPopup(auth, provider)  Google login popup
  signOut(auth)                    Logout
  onAuthStateChanged(auth, cb)     Listen to auth state changes
